prototype.item_update=function(e){
  if(e){
    var t=this.getHouseItemCount(e.item.item_id);
    0==e.item.count?this.doConsumeHouseItem(e.item.item_id,t):this.doAddHouseItem(e.item.item_id,e.item.count-t);
    var i=Tabikaeru.DataManager.instance().ItemDB.get(e.item.item_id);
    if(i.type==Tabikaeru.DataType.ItemType.COMPOSE){
      core.ModelManage.getInstance().getModel(HandCraftModel).updateComposeRedot();
      var n=core.PageManage.getInstance().getControl(ToolBagViewController,core.ViewLayerType.NoticeLayer);
      n&&n.view.updatePage()
    }
  }
}
